let rocketX = 0; let rocketY = 0; // دالة تحديث شريط آخر جولات على الشاشة function updateHistoryUI() { historyBar.innerHTML = ''; // عرض آخر 5 جولات فقط لمنع الازدحام const lastFive = historyList.slice(-5).reverse(); lastFive.forEach(val => { const item = document.createElement('div'); item.className = 'history-item'; const num = parseFloat(val); if (num <= 1.2) item.classList.add('crash-low'); else if (num < 5) item.classList.add('crash-med'); else item.classList.add('crash-high'); item.innerText = val; historyBar.appendChild(item); }); } updateHistoryUI(); // حلقة الرسم والتحديث المستمر (الخلفية والخط والصاروخ) function draw() { ctx.clearRect(0, 0, canvas.width, canvas.height); // 1. رسم وتحديث النجوم في الخلفية (تحرك مائل لأسفل يعطي إيحاء صعود الصاروخ لأعلى) ctx.fillStyle = '#ffffff'; stars.forEach(star => { ctx.beginPath(); ctx.arc(star.x, star.y, star.size, 0, Math.PI * 2); ctx.fill(); if (gameState === 'FLYING') { star.x -= star.speed * 2; star.y += star.speed * 2; } else { star.x -= star.speed * 0.2; star.y += star.speed * 0.2; } if (star.x < 0) star.x = canvas.width; if (star.y > canvas.height) star.y = 0; }); // 2. رسم مسار الخط النيوني المتوهج خلف الصاروخ if (flightPath.length > 1) { ctx.beginPath(); ctx.moveTo(flightPath[0].x, flightPath[0].y); for (let i = 1; i < flightPath.length; i++) { ctx.lineTo(flightPath[i].x, flightPath[i].y); } ctx.strokeStyle = '#66fcf1'; ctx.lineWidth = 4; ctx.shadowBlur = 15; ctx.shadowColor = '#66fcf1'; ctx.stroke(); ctx.shadowBlur = 0; // إعادة تصفير التوهج حتى لا يخرب باقي العناصر } // 3. رسم وحركة الصاروخ بشكل منحنٍ للزاوية العلوية if (gameState === 'FLYING') { multiplier += 0.008; // سرعة حساب المضاعف multiplierDisplay.innerText = multiplier.toFixed(2) + 'x'; rocketY = (canvas.height - 40) - (canvas.height - 120) * Math.sin(progress * Math.PI / 2); flightPath.push({x: rocketX, y: rocketY}); // رسم مجسم الصاروخ (نص إيموجي مائل بزاوية طيران) ctx.save(); ctx.translate(rocketX, rocketY); ctx.rotate(-Math.PI / 4); // إمالة الصاروخ للزاوية عاليًا ctx.font = '38px Arial'; ctx.textAlign = 'center'; ctx.textBaseline = 'middle'; ctx.fillText('🚀', 0, 0); ctx.restore(); // التحقق من نقطة الانفجار اللحظية if (multiplier >= crashPoint) { endGame(false); } } else if (gameState === 'CRASHED') { // رسم تأثير الانفجار مكان توقف الصاروخ } else if (gameState === 'CRASHED') { ctx.font = '45px Arial'; ctx.textAlign = 'center'; ctx.textBaseline = 'middle'; ctx.fillText('💥', rocketX, rocketY); } else { // وضع الانتظار الأولي للصاروخ أسفل اليسار جاهز للإطلاق rocketX = 40; rocketY = canvas.height - 40; ctx.save(); ctx.translate(rocketX, rocketY); ctx.rotate(-Math.PI / 4); ctx.font = '38px Arial'; ctx.textAlign = 'center'; ctx.textBaseline = 'middle'; ctx.fillText('🚀', 0, 0); ctx.restore(); } gameLoopId = requestAnimationFrame(draw); } // بدء حلقة الرسوم المتحركة تلقائياً عند التشغيل لأول مرة draw();
